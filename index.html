DATABASE NORMALIZATION

Normalized Schema

Designed to reduce duplication and maintain data integrity through 1NF, 2NF, and 3NF.

01

First Normal Form (1NF)

Each field stores a single atomic value. Repeating groups are avoided, and each table has a primary key.

  • Atomic student and profile attributes
  • One fee per student-fee record
  • One payment item per payment-fee combination
02

Second Normal Form (2NF)

Non-key attributes depend on the complete primary key rather than only part of a key.

  • Payment details belong to payments
  • Fee definitions belong to fee_types
  • Student-specific fee balances belong to student_fees
03

Third Normal Form (3NF)

Non-key attributes depend on the key, the whole key, and nothing but the key.

  • User information is separated into profiles
  • Fee descriptions are stored once in fee_types
  • Payment line items are separated from payment headers

DATABASE IMPLEMENTATION

SQL Script

-- SCHOOL PAYMENT SYSTEM
-- PostgreSQL / Supabase compatible

CREATE TYPE user_role AS ENUM ('student', 'cashier', 'admin');
CREATE TYPE payment_method AS ENUM ('cash', 'gcash');
CREATE TYPE payment_status AS ENUM ('pending', 'successful', 'failed', 'cancelled');

CREATE TABLE profiles (
    id uuid PRIMARY KEY,
    full_name text NOT NULL,
    email text UNIQUE,
    role user_role NOT NULL,
    created_at timestamptz DEFAULT now()
);

CREATE TABLE students (
    student_id text PRIMARY KEY,
    profile_id uuid UNIQUE REFERENCES profiles(id),
    first_name text NOT NULL,
    middle_name text,
    last_name text NOT NULL,
    course text,
    year_level int,
    section text,
    contact_number text,
    created_at timestamptz DEFAULT now()
);

CREATE TABLE fee_types (
    fee_type_id bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    fee_name text NOT NULL UNIQUE,
    description text,
    created_at timestamptz DEFAULT now()
);

CREATE TABLE student_fees (
    student_fee_id bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    student_id text NOT NULL REFERENCES students(student_id),
    fee_type_id bigint NOT NULL REFERENCES fee_types(fee_type_id),
    amount_due decimal(12,2) NOT NULL CHECK (amount_due >= 0),
    amount_paid decimal(12,2) NOT NULL DEFAULT 0 CHECK (amount_paid >= 0),
    due_date date,
    created_at timestamptz DEFAULT now(),
    CHECK (amount_paid <= amount_due)
);

CREATE TABLE payments (
    payment_id bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    receipt_number text NOT NULL UNIQUE,
    student_id text NOT NULL REFERENCES students(student_id),
    cashier_id uuid REFERENCES profiles(id),
    payment_method payment_method NOT NULL,
    total_amount decimal(12,2) NOT NULL CHECK (total_amount > 0),
    status payment_status DEFAULT 'pending',
    payment_date timestamptz DEFAULT now(),
    notes text
);

CREATE TABLE payment_items (
    payment_item_id bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    payment_id bigint NOT NULL REFERENCES payments(payment_id),
    student_fee_id bigint NOT NULL REFERENCES student_fees(student_fee_id),
    amount_paid decimal(12,2) NOT NULL CHECK (amount_paid > 0)
);

CREATE INDEX idx_student_fees_student ON student_fees(student_id);
CREATE INDEX idx_student_fees_type ON student_fees(fee_type_id);
CREATE INDEX idx_payments_student ON payments(student_id);
CREATE INDEX idx_payments_cashier ON payments(cashier_id);
CREATE INDEX idx_payment_items_payment ON payment_items(payment_id);
CREATE INDEX idx_payment_items_fee ON payment_items(student_fee_id);
